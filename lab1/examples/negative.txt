# Синтаксические ошибки: соседние буквы, скобка, пропущенная операция.
mn -> (p -> p)
p -> (p -> p))
q -> (p p)
# Синтаксически корректны, но одного разрешенного шага нет.
p
(a | a) -> b
# Нарушено одинаковое значение p в схеме A1.
a -> (c | b)
# Частичная замена p в A1: первое вхождение p изменено, второе нет.
a -> (p | q)
# Коммутативность не выполняется автоматически.
p | ~p
